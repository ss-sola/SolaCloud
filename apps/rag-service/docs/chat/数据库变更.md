-- Deep Agent 运行时数据库表
-- 用于管理 Deep Agents 的沙箱工作区和命名空间隔离

-- ============================================================
-- 一、业务表（手动创建）
-- ============================================================

-- 1. 命名空间表：隔离不同业务线或租户的 Agent 运行时数据
CREATE TABLE IF NOT EXISTS deep_agent_namespace (
id SERIAL PRIMARY KEY,
namespace_code VARCHAR(128) NOT NULL UNIQUE,
display_name VARCHAR(255),
description TEXT,
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
deleted_at TIMESTAMP,
creator_id INT,
editor_id INT
);

COMMENT ON TABLE deep_agent_namespace IS 'Deep Agent 命名空间';
COMMENT ON COLUMN deep_agent_namespace.namespace_code IS '命名空间唯一标识';
COMMENT ON COLUMN deep_agent_namespace.display_name IS '命名空间显示名称';
COMMENT ON COLUMN deep_agent_namespace.description IS '命名空间描述';

-- 2. 工作区表：每个 workspace 对应一个沙箱后端实例及其运行时上下文
CREATE TABLE IF NOT EXISTS deep_agent_workspace (
id SERIAL PRIMARY KEY,
namespace_id INT NOT NULL REFERENCES deep_agent_namespace(id),
workspace_id VARCHAR(128) NOT NULL UNIQUE,
display_name VARCHAR(255),
thread_id VARCHAR(255),
run_id VARCHAR(255),
session_id INT,
account_id INT NOT NULL,
status VARCHAR(32) NOT NULL DEFAULT 'running',
root_path VARCHAR(512),
disk_quota_mb INT,
last_active_at TIMESTAMP,
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
deleted_at TIMESTAMP,
creator_id INT,
editor_id INT
);

COMMENT ON TABLE deep_agent_workspace IS 'Deep Agent 工作区';
COMMENT ON COLUMN deep_agent_workspace.namespace_id IS '关联的命名空间 ID';
COMMENT ON COLUMN deep_agent_workspace.workspace_id IS '工作区唯一标识，用于沙箱后端路由';
COMMENT ON COLUMN deep_agent_workspace.thread_id IS 'LangGraph 线程 ID';
COMMENT ON COLUMN deep_agent_workspace.run_id IS 'LangGraph run ID';
COMMENT ON COLUMN deep_agent_workspace.session_id IS '关联的聊天会话 ID';
COMMENT ON COLUMN deep_agent_workspace.account_id IS '关联的账号 ID';
COMMENT ON COLUMN deep_agent_workspace.status IS '工作区状态：running/completed/failed/cancelled';
COMMENT ON COLUMN deep_agent_workspace.root_path IS '工作区根目录路径（沙箱内虚拟路径）';
COMMENT ON COLUMN deep_agent_workspace.disk_quota_mb IS '工作区创建时的磁盘配额（MB）';
COMMENT ON COLUMN deep_agent_workspace.last_active_at IS '工作区最后一次活跃时间';

-- 索引
CREATE INDEX IF NOT EXISTS idx_deep_agent_workspace_namespace_id ON deep_agent_workspace(namespace_id);
CREATE INDEX IF NOT EXISTS idx_deep_agent_workspace_session_id ON deep_agent_workspace(session_id);
CREATE INDEX IF NOT EXISTS idx_deep_agent_workspace_account_id ON deep_agent_workspace(account_id);
CREATE INDEX IF NOT EXISTS idx_deep_agent_workspace_status ON deep_agent_workspace(status) WHERE deleted_at IS NULL;

-- 3. 插入默认命名空间（幂等）
INSERT INTO deep_agent_namespace (namespace_code, display_name, description, created_at, updated_at)
VALUES ('default', '默认命名空间', '系统默认 Deep Agent 命名空间', CURRENT_TIMESTAMP, CURRENT_TIMESTAMP)
ON CONFLICT (namespace_code) DO NOTHING;

-- 二、LangGraph 持久化表（自动创建）
-- ============================================================
-- 以下表由 PostgresSaver.setup() 和 PostgresStore.start() 在首次初始化时自动创建，
-- 无需手动执行 DDL。

-- langgraph_checkpoints — 存储 checkpoint（对话状态快照、消息历史）
-- langgraph_blobs — 存储 blob 数据
-- langgraph_metadata — 存储 checkpoint 元数据

-- PostgresStore 自动创建的表：
-- langgraph_store — 存储跨线程共享数据（namespace/key/value）

-- 注意：PostgresSaver.setup() 和 PostgresStore.start() 是幂等的，
-- 重复调用不会产生错误。
